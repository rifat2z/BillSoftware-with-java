import javax.swing.*;
import javax.swing.event.TableModelEvent;
import javax.swing.table.*;
import java.awt.*;
import java.awt.event.*;
import java.awt.print.*;
import java.io.*;
import java.text.DecimalFormat;
import java.time.LocalDateTime;
import javax.imageio.ImageIO;
import java.awt.image.BufferedImage;

public class BillingApp extends JFrame {
    private static final String SHOP_NAME = "BONDHON PHOTOSTAT & STUDIO";
    private static final String SHOP_ADDRESS = "Khander, Gohail Road, Bogura - Phone: 01714-928011";
    private static final String EMAIL_STRING = "khanphotostat701@gmail.com";
    private static final DecimalFormat DF = new DecimalFormat("#,##0.00");

    private DefaultTableModel tableModel;
    private JTable table;
    private JLabel subtotalLabel, taxLabel, discountLabel, totalLabel;
    private JTextField taxField, discountField, customerField;

    // Auto Invoice Number
    private int invoiceNumber = 1;
    private final String INVOICE_COUNTER_FILE = "invoice_counter.txt";

    // Logo
    private BufferedImage logoImage;

    public BillingApp() {
        super(SHOP_NAME + " - Billing Software");
        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        setSize(950, 650);
        setLocationRelativeTo(null);

        loadInvoiceNumber();
        loadLogo();
        initUI();
    }

    private void loadLogo() {
        try {
            logoImage = ImageIO.read(new File("rifat.png"));
        } catch (Exception e) {
            logoImage = null;
        }
    }

    private void loadInvoiceNumber() {
        File f = new File(INVOICE_COUNTER_FILE);
        if (f.exists()) {
            try (BufferedReader br = new BufferedReader(new FileReader(f))) {
                String line = br.readLine();
                if (line != null) invoiceNumber = Integer.parseInt(line.trim());
            } catch (Exception ex) {
                invoiceNumber = 1;
            }
        }
    }

    private void saveInvoiceNumber() {
        try (PrintWriter pw = new PrintWriter(new FileWriter(INVOICE_COUNTER_FILE))) {
            pw.println(invoiceNumber);
        } catch (Exception ex) {}
    }

    private void initUI() {
        JPanel header = new JPanel(new BorderLayout());
        JLabel title = new JLabel("<html><div style='text-align:center'><h2>" + SHOP_NAME + "</h2><small>" + SHOP_ADDRESS + "</small></div></html>", SwingConstants.CENTER);
        header.add(title, BorderLayout.CENTER);
        add(header, BorderLayout.NORTH);

        String[] cols = {"Description", "Qty", "Unit Price", "Total"};
        tableModel = new DefaultTableModel(cols, 0) {
            @Override
            public boolean isCellEditable(int row, int column) {
                return column != 3;
            }
        };
        table = new JTable(tableModel);
        table.setRowHeight(24);
        JScrollPane sp = new JScrollPane(table);
        add(sp, BorderLayout.CENTER);

        JPanel right = new JPanel();
        right.setLayout(new BoxLayout(right, BoxLayout.Y_AXIS));
        right.setBorder(BorderFactory.createEmptyBorder(10,10,10,10));
        right.add(new JLabel("Customer Name:"));
        customerField = new JTextField();
        right.add(customerField);
        right.add(Box.createRigidArea(new Dimension(0,10)));

        JPanel calcPanel = new JPanel(new GridLayout(0,2,4,4));
        calcPanel.add(new JLabel("Tax %:"));
        taxField = new JTextField("0.00");
        calcPanel.add(taxField);
        calcPanel.add(new JLabel("Discount:"));
        discountField = new JTextField("0.00");
        calcPanel.add(discountField);
        right.add(calcPanel);
        right.add(Box.createRigidArea(new Dimension(0,10)));

        subtotalLabel = new JLabel("Subtotal: 0.00");
        taxLabel = new JLabel("Tax: 0.00");
        discountLabel = new JLabel("Discount: 0.00");
        totalLabel = new JLabel("<html><b>Total: 0.00</b></html>");
        right.add(subtotalLabel);
        right.add(taxLabel);
        right.add(discountLabel);
        right.add(totalLabel);
        right.add(Box.createRigidArea(new Dimension(0,10)));

        JButton addBtn = new JButton("Add Item");
        addBtn.addActionListener(e -> showAddItemDialog());
        JButton removeBtn = new JButton("Remove Selected");
        removeBtn.addActionListener(e -> removeSelectedRows());
        JButton calcBtn = new JButton("Recalculate");
        calcBtn.addActionListener(e -> recalculateTotals());
        JButton printPreviewBtn = new JButton("Print Preview");
        printPreviewBtn.addActionListener(e -> printPreview());
        JButton directPrintBtn = new JButton("Print (Direct)");
        directPrintBtn.addActionListener(e -> directPrint());
        JButton saveBtn = new JButton("Save Invoice (CSV)");
        saveBtn.addActionListener(e -> saveInvoiceCSV());
        JButton saveTextBtn = new JButton("Save Invoice (Text)");
        saveTextBtn.addActionListener(e -> saveInvoiceText());
        JButton clearBtn = new JButton("Clear All");
        clearBtn.addActionListener(e -> clearAll());

        right.add(addBtn);
        right.add(removeBtn);
        right.add(calcBtn);
        right.add(printPreviewBtn);
        right.add(directPrintBtn);
        right.add(saveBtn);
        right.add(saveTextBtn);
        right.add(clearBtn);

        add(right, BorderLayout.EAST);

        JPanel bottom = new JPanel(new FlowLayout(FlowLayout.LEFT));
        bottom.add(new JLabel("Tip: Double-click Qty or Unit Price to edit. Then press 'Recalculate'."));
        add(bottom, BorderLayout.SOUTH);

        setUpTableEditors();
        tableModel.addTableModelListener(e -> recalcRowTotal(e));
    }

    private void setUpTableEditors() {
        TableColumnModel colModel = table.getColumnModel();
        colModel.getColumn(1).setPreferredWidth(60);
        colModel.getColumn(2).setPreferredWidth(80);
        colModel.getColumn(0).setPreferredWidth(300);
        colModel.getColumn(3).setPreferredWidth(100);
    }

    private void showAddItemDialog() {
        JTextField desc = new JTextField();
        JTextField qty = new JTextField("1");
        JTextField price = new JTextField("0.00");

        JPanel p = new JPanel(new GridLayout(0,1));
        p.add(new JLabel("Description:"));
        p.add(desc);
        p.add(new JLabel("Quantity:"));
        p.add(qty);
        p.add(new JLabel("Unit Price:"));
        p.add(price);

        int res = JOptionPane.showConfirmDialog(this, p, "Add Item", JOptionPane.OK_CANCEL_OPTION, JOptionPane.PLAIN_MESSAGE);
        if (res == JOptionPane.OK_OPTION) {
            String d = desc.getText().trim();
            int q = parseIntSafe(qty.getText().trim(), 1);
            double pr = parseDoubleSafe(price.getText().trim(), 0.0);
            double total = q * pr;
            tableModel.addRow(new Object[] { d, q, DF.format(pr), DF.format(total) });
            recalculateTotals();
        }
    }

    private int parseIntSafe(String s, int def) {
        try { return Integer.parseInt(s); } catch (Exception ex) { return def; }
    }

    private double parseDoubleSafe(String s, double def) {
        try { s = s.replaceAll(",", ""); return Double.parseDouble(s); } catch (Exception ex) { return def; }
    }

    private void removeSelectedRows() {
        int[] rows = table.getSelectedRows();
        if (rows.length == 0) { JOptionPane.showMessageDialog(this, "Select one or more rows to remove."); return; }
        for (int i = rows.length -1; i >= 0; i--) tableModel.removeRow(rows[i]);
        recalculateTotals();
    }

    private void recalcRowTotal(TableModelEvent e) {
        int row = e.getFirstRow();
        int col = e.getColumn();
        if (row < 0) return;
        if (col == 1 || col == 2 || col == TableModelEvent.ALL_COLUMNS) {
            try {
                int qty = parseIntSafe("" + tableModel.getValueAt(row,1),1);
                double price = parseDoubleSafe("" + tableModel.getValueAt(row,2),0.0);
                double total = qty * price;
                tableModel.setValueAt(DF.format(total), row, 3);
            } catch (Exception ex) {}
            recalculateTotals();
        }
    }

    private void recalculateTotals() {
        double subtotal=0;
        for(int i=0;i<tableModel.getRowCount();i++){
            subtotal+=parseDoubleSafe("" + tableModel.getValueAt(i,3),0.0);
        }
        double taxPct = parseDoubleSafe(taxField.getText().trim(),0.0);
        double discountAmount = parseDoubleSafe(discountField.getText().trim(),0.0);

        double taxAmount = subtotal * taxPct /100.0;
        double finalTotal = subtotal + taxAmount - discountAmount;
        if(finalTotal<0) finalTotal = 0.0;

        subtotalLabel.setText("Subtotal: "+DF.format(subtotal));
        taxLabel.setText("Tax: "+DF.format(taxAmount));
        discountLabel.setText("Discount: "+DF.format(discountAmount));
        totalLabel.setText("<html><b>Total: "+DF.format(finalTotal)+"</b></html>");
    }

    private String buildInvoiceText() {
        StringBuilder sb = new StringBuilder();
        sb.append("Invoice #: INV-"+String.format("%03d",invoiceNumber)).append("\n");
        sb.append("Customer: ").append(customerField.getText().trim()).append("\n");
        sb.append("Date: ").append(LocalDateTime.now()).append("\n\n");

        sb.append(String.format("%-28s %5s %8s %8s\n","Description","Qty","Price","Total"));
        sb.append("------------------------------------------------\n");

        double subtotal=0.0;
        for(int i=0;i<tableModel.getRowCount();i++){
            String desc=""+tableModel.getValueAt(i,0);
            String qty=""+tableModel.getValueAt(i,1);
            String price=""+tableModel.getValueAt(i,2);
            String tot=""+tableModel.getValueAt(i,3);
            subtotal+=parseDoubleSafe(tot,0.0);
            if(desc.length()>28) desc=desc.substring(0,28);
            sb.append(String.format("%-28s %5s %8s %8s\n",desc,qty,price,tot));
        }

        double taxPct = parseDoubleSafe(taxField.getText().trim(),0.0);
        double discountAmount = parseDoubleSafe(discountField.getText().trim(),0.0);
        double taxAmount = subtotal*taxPct/100.0;
        double finalTotal = subtotal+taxAmount-discountAmount;

        sb.append("------------------------------------------------\n");
        sb.append(String.format("%-36s %10s\n","Subtotal:",DF.format(subtotal)));
        sb.append(String.format("%-36s %10s\n","Tax ("+DF.format(taxPct)+"%):",DF.format(taxAmount)));
        sb.append(String.format("%-36s %10s\n","Discount:",DF.format(discountAmount)));
        sb.append("------------------------------------------------\n");
        sb.append(String.format("%-36s %10s\n","Total:",DF.format(finalTotal)));
        sb.append("------------------------------------------------\n");
        sb.append("Thank you for your business!\n");
        sb.append("Contact: WhatsApp-01714-928011 E-mail: "+EMAIL_STRING+"\n");

        invoiceNumber++;
        saveInvoiceNumber();
        return sb.toString();
    }

    private void printPreview() {
        JTextArea ta = new JTextArea(buildInvoiceText());
        ta.setFont(new Font(Font.MONOSPACED,Font.PLAIN,12));
        ta.setEditable(false);
        JScrollPane sp = new JScrollPane(ta);
        sp.setPreferredSize(new Dimension(600,500));

        // Logo Preview
        if(logoImage!=null){
            JLabel logoLabel = new JLabel(new ImageIcon(logoImage.getScaledInstance(120,80,Image.SCALE_SMOOTH)));
            JPanel panel = new JPanel(new BorderLayout());
            panel.add(logoLabel, BorderLayout.NORTH);
            panel.add(sp, BorderLayout.CENTER);
            JOptionPane.showMessageDialog(this,panel,"Invoice Preview",JOptionPane.PLAIN_MESSAGE);
        } else {
            JOptionPane.showMessageDialog(this,sp,"Invoice Preview",JOptionPane.PLAIN_MESSAGE);
        }
    }

    private void directPrint() {
        PrinterJob job = PrinterJob.getPrinterJob();
        job.setJobName("Invoice Print");

        job.setPrintable((graphics, pageFormat, pageIndex) -> {
            if(pageIndex>0) return Printable.NO_SUCH_PAGE;
            Graphics2D g2d = (Graphics2D) graphics;
            g2d.translate(pageFormat.getImageableX(), pageFormat.getImageableY());

            int y = 20;
            if(logoImage!=null){
                g2d.drawImage(logoImage, 0, y, 120,80,null);
                y+=90;
            }

            g2d.setFont(new Font(Font.MONOSPACED,Font.PLAIN,12));
            for(String line : buildInvoiceText().split("\n")){
                g2d.drawString(line,0,y);
                y+=15;
            }
            return Printable.PAGE_EXISTS;
        });

        boolean ok = job.printDialog();
        if(ok){
            try{job.print(); JOptionPane.showMessageDialog(this,"Invoice sent to printer.");}
            catch(PrinterException ex){JOptionPane.showMessageDialog(this,"Printing failed: "+ex.getMessage());}
        }
    }

    private void saveInvoiceCSV(){
        JFileChooser fc = new JFileChooser();
        fc.setSelectedFile(new File("invoice.csv"));
        int res = fc.showSaveDialog(this);
        if(res!=JFileChooser.APPROVE_OPTION) return;
        File f=fc.getSelectedFile();
        try(PrintWriter pw=new PrintWriter(new FileWriter(f))){
            pw.println("Description,Qty,UnitPrice,Total");
            for(int i=0;i<tableModel.getRowCount();i++){
                pw.println(tableModel.getValueAt(i,0)+","+tableModel.getValueAt(i,1)+","+tableModel.getValueAt(i,2)+","+tableModel.getValueAt(i,3));
            }
            JOptionPane.showMessageDialog(this,"Saved CSV to: "+f.getAbsolutePath());
        }catch(Exception ex){JOptionPane.showMessageDialog(this,"Error saving CSV: "+ex.getMessage());}
    }

    private void saveInvoiceText(){
        JFileChooser fc = new JFileChooser();
        fc.setSelectedFile(new File("invoice.txt"));
        int res = fc.showSaveDialog(this);
        if(res!=JFileChooser.APPROVE_OPTION) return;
        File f=fc.getSelectedFile();
        try(PrintWriter pw=new PrintWriter(new FileWriter(f))){
            pw.println(buildInvoiceText());
            JOptionPane.showMessageDialog(this,"Saved invoice text to: "+f.getAbsolutePath());
        }catch(Exception ex){JOptionPane.showMessageDialog(this,"Error saving text: "+ex.getMessage());}
    }

    private void clearAll(){
        tableModel.setRowCount(0);
        customerField.setText("");
        taxField.setText("0.00");
        discountField.setText("0.00");
        recalculateTotals();
    }

    public static void main(String[] args){
        SwingUtilities.invokeLater(() -> {
            BillingApp app = new BillingApp();
            app.setVisible(true);
        });
    }
}
